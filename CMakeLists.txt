cmake_minimum_required(VERSION 3.25)
project(SerikaPhotoEdit VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
option(SERIKA_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(SERIKA_WITH_RAW "Enable dynamically linked LibRaw" ON)
option(SERIKA_WITH_GPU "Enable optional OpenGL image processing" ON)
option(SERIKA_REQUIRE_CMYK "Require LittleCMS for ICC CMYK proofing and export" OFF)
option(SERIKA_ALPINE "Allow an Alpine/musl build" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network PrintSupport Svg Test)
find_package(Qt6 6.8 QUIET COMPONENTS Pdf)
find_package(ZLIB QUIET)
find_package(PkgConfig QUIET)
if(SERIKA_WITH_GPU)
  find_package(Qt6 6.8 QUIET COMPONENTS OpenGL)
endif()
find_package(lcms2 CONFIG QUIET)
if(NOT TARGET lcms2::lcms2 AND PkgConfig_FOUND)
  pkg_check_modules(LCMS2 QUIET IMPORTED_TARGET lcms2)
endif()
if(NOT TARGET lcms2::lcms2 AND NOT TARGET PkgConfig::LCMS2)
  find_path(LCMS2_INCLUDE_DIR lcms2.h)
  find_library(LCMS2_LIBRARY NAMES lcms2 liblcms2)
endif()
if(SERIKA_REQUIRE_CMYK AND NOT TARGET lcms2::lcms2 AND NOT TARGET PkgConfig::LCMS2 AND NOT (LCMS2_INCLUDE_DIR AND LCMS2_LIBRARY))
  message(FATAL_ERROR "LittleCMS2 is required. Install liblcms2-dev / little-cms2, or use scripts/build-lcms.ps1.")
endif()
if(SERIKA_WITH_RAW)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIBRAW QUIET IMPORTED_TARGET libraw)
  endif()
  if(NOT LIBRAW_FOUND)
    find_path(LIBRAW_INCLUDE_DIR libraw/libraw.h)
    find_library(LIBRAW_LIBRARY NAMES raw libraw)
    if(LIBRAW_INCLUDE_DIR AND LIBRAW_LIBRARY)
      set(LIBRAW_FOUND TRUE)
    endif()
  endif()
endif()
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/document/*.cpp src/compositor/*.cpp src/io/*.cpp src/filters/*.cpp src/actions/*.cpp)
# The version lives only in project(VERSION ...); code reads it from the generated header.
configure_file(cmake/Version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/serika/Version.h @ONLY)
add_library(serika_engine STATIC ${ENGINE_SOURCES} src/document/Document.h src/io/FormatIO.h)
target_include_directories(serika_engine PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(serika_engine PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets)
if(TARGET Qt6::OpenGL AND SERIKA_WITH_GPU)
  target_link_libraries(serika_engine PRIVATE Qt6::OpenGL)
  target_compile_definitions(serika_engine PRIVATE SERIKA_HAVE_OPENGL=1)
endif()
if(TARGET lcms2::lcms2)
  target_link_libraries(serika_engine PUBLIC lcms2::lcms2)
  target_compile_definitions(serika_engine PUBLIC SERIKA_HAVE_LCMS2=1)
elseif(TARGET PkgConfig::LCMS2)
  target_link_libraries(serika_engine PUBLIC PkgConfig::LCMS2)
  target_compile_definitions(serika_engine PUBLIC SERIKA_HAVE_LCMS2=1)
elseif(LCMS2_INCLUDE_DIR AND LCMS2_LIBRARY)
  target_include_directories(serika_engine PUBLIC ${LCMS2_INCLUDE_DIR})
  target_link_libraries(serika_engine PUBLIC ${LCMS2_LIBRARY})
  target_compile_definitions(serika_engine PUBLIC SERIKA_HAVE_LCMS2=1)
endif()
if(TARGET Qt6::Pdf)
  target_link_libraries(serika_engine PRIVATE Qt6::Pdf)
  target_compile_definitions(serika_engine PRIVATE SERIKA_HAVE_QTPDF=1)
endif()
if(ZLIB_FOUND)
  target_link_libraries(serika_engine PRIVATE ZLIB::ZLIB)
  target_compile_definitions(serika_engine PRIVATE SERIKA_HAVE_ZLIB=1)
endif()
if(LIBRAW_FOUND)
  if(TARGET PkgConfig::LIBRAW)
    target_link_libraries(serika_engine PRIVATE PkgConfig::LIBRAW)
  else()
    target_include_directories(serika_engine PRIVATE ${LIBRAW_INCLUDE_DIR})
    target_link_libraries(serika_engine PRIVATE ${LIBRAW_LIBRARY})
  endif()
  target_compile_definitions(serika_engine PRIVATE SERIKA_HAVE_LIBRAW=1)
endif()
file(GLOB_RECURSE UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/tools/*.cpp)
add_library(serika_ui STATIC ${UI_SOURCES} src/ui/canvas/CanvasView.h src/ui/MainWindow.h)
target_link_libraries(serika_ui PUBLIC serika_engine Qt6::PrintSupport Qt6::Svg)
# ONNX Runtime is optional and loaded at runtime; only its MIT C API header is vendored.
target_include_directories(serika_ui SYSTEM PRIVATE third_party/onnxruntime/include)
target_link_libraries(serika_ui PRIVATE Qt6::Network)
add_library(serika_mcp STATIC src/mcp/McpServer.cpp src/mcp/McpLiveBridge.cpp)
target_link_libraries(serika_mcp PUBLIC serika_ui Qt6::Network)
qt_add_executable(SerikaPhotoEdit WIN32 MACOSX_BUNDLE src/app/main.cpp resources/serika.qrc)
target_link_libraries(SerikaPhotoEdit PRIVATE serika_ui serika_mcp)
if(WIN32)
  target_link_libraries(SerikaPhotoEdit PRIVATE dwmapi shell32 ole32)
  set_source_files_properties(packaging/windows/SerikaPhotoEdit.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/serika.ico")
  target_sources(SerikaPhotoEdit PRIVATE packaging/windows/SerikaPhotoEdit.rc)
endif()
if(APPLE)
  set_source_files_properties(packaging/macos/SerikaPhotoEdit.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(SerikaPhotoEdit PRIVATE packaging/macos/SerikaPhotoEdit.icns)
endif()
foreach(target serika_engine serika_ui serika_mcp SerikaPhotoEdit)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
    if(SERIKA_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(SERIKA_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endforeach()
set_target_properties(SerikaPhotoEdit PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Serika PhotoEdit" MACOSX_BUNDLE_GUI_IDENTIFIER "io.serika.PhotoEdit" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist" MACOSX_BUNDLE_ICON_FILE "SerikaPhotoEdit.icns")
if(BUILD_TESTING)
  if(WIN32)
    # Qt's offscreen platform uses a file font database instead of Windows font discovery.
    file(TO_CMAKE_PATH "$ENV{SystemRoot}/Fonts" serika_test_font_dir)
  endif()
  foreach(suite core io ui canvas actions shortcuts mask workflow curves fills brush layer_operations smartobject gpu color_management typography mcp)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_tests.cpp")
      qt_add_executable(${suite}_tests tests/${suite}_tests.cpp resources/serika.qrc)
      target_link_libraries(${suite}_tests PRIVATE serika_ui Qt6::Test)
      if(suite STREQUAL "mcp")
        target_link_libraries(${suite}_tests PRIVATE serika_mcp)
        target_compile_definitions(${suite}_tests PRIVATE SERIKA_APP_PATH="$<TARGET_FILE:SerikaPhotoEdit>")
        add_dependencies(${suite}_tests SerikaPhotoEdit)
      endif()
      add_test(NAME ${suite} COMMAND ${suite}_tests -platform offscreen
        -o ${CMAKE_CURRENT_BINARY_DIR}/${suite}-results.txt,txt -o -,txt)
      set_tests_properties(${suite} PROPERTIES TIMEOUT 90)
      if(WIN32)
        set_property(TEST ${suite} APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=${serika_test_font_dir}")
      endif()
    endif()
  endforeach()
  qt_add_executable(performance_bench tests/performance_bench.cpp)
  target_link_libraries(performance_bench PRIVATE serika_engine)
endif()
install(TARGETS SerikaPhotoEdit BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE README.md CONTRIBUTING.md CHANGELOG.md docs/THIRD-PARTY.md DESTINATION share/serika-photoedit)
install(DIRECTORY docs/ DESTINATION share/serika-photoedit/docs FILES_MATCHING PATTERN "*.md" PATTERN "*.png" PATTERN "*.json")
install(FILES src/io/README.md DESTINATION share/serika-photoedit/src/io)
install(DIRECTORY examples/ DESTINATION share/serika-photoedit/examples)
install(DIRECTORY resources/licenses/ DESTINATION share/serika-photoedit/licenses)
if(WIN32)
  if(TARGET lcms2::lcms2)
    install(FILES $<TARGET_FILE:lcms2::lcms2> DESTINATION bin)
  endif()
  # The headless --batch entry point uses Qt's offscreen platform plug-in.
  get_target_property(qt_core_dll Qt6::Core IMPORTED_LOCATION_RELEASE)
  get_filename_component(qt_runtime_dir "${qt_core_dll}" DIRECTORY)
  if(EXISTS "${qt_runtime_dir}/../plugins/platforms/qoffscreen.dll")
    install(FILES "${qt_runtime_dir}/../plugins/platforms/qoffscreen.dll" DESTINATION plugins/platforms)
  endif()
  foreach(dependency LIBRAW_LIBRARY ZLIB_LIBRARY LCMS2_LIBRARY)
    if(${dependency})
      get_filename_component(dependency_lib_dir "${${dependency}}" DIRECTORY)
      foreach(dll libraw.dll libzlib.dll zlib1.dll liblcms2.dll lcms2.dll)
        if(EXISTS "${dependency_lib_dir}/../bin/${dll}")
          install(FILES "${dependency_lib_dir}/../bin/${dll}" DESTINATION bin)
        endif()
      endforeach()
    endif()
  endforeach()
endif()
install(DIRECTORY resources/icons DESTINATION share/serika-photoedit)
if(UNIX AND NOT APPLE)
  install(FILES packaging/linux/io.serika.PhotoEdit.desktop DESTINATION share/applications)
  install(FILES packaging/linux/io.serika.PhotoEdit.xml DESTINATION share/mime/packages)
  install(FILES resources/icons/serika-photoedit.png DESTINATION share/icons/hicolor/512x512/apps)
endif()
qt_generate_deploy_app_script(TARGET SerikaPhotoEdit OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
include(cmake/Packaging.cmake)
